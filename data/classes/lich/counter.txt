{
	flags: [ persist_in_death ]	
	type: hidden
	aura_effect combat_event_trigger: 
	{ 
		combat_event_ability_mask: [ melee ]
		combat_event<source>: [ dodge parry ]
		ability<target>: $ability
		{
			flags: [ always_in_range always_in_line_of_sight target_hostile target_other ]
			states: [ default in_combat ]
			icon: icon_kick_2
			delay: 2				
			direct_effect apply_aura:
			{
				aura: $aura @counterspellkick_trigger
				{
					string: "Exposed"
					description: "Dodged or parried one of your attacks."
					icon: icon_kick_2
					flags: [ ]
					type: debuff
					duration: 40
				}
			}		
		}
	}
}
